import pandas as pd
        import matplotlib.pyplot as plt
        import seaborn as sns
        data = sns.load_dataset('tips')

        plt.scatter(data['day'], data['tip'], c=data['size'], 
                    s=data['total_bill'])


        plt.title("Scatter Plot")
        plt.xlabel('Day')
        plt.ylabel('Tip')
        plt.colorbar()
        plt.show()
