        import sys

        current_word, total = None, 0

        for line in sys.stdin:
            word, count = line.strip().split('\t')
            count = int(count)
            if current_word == word:
                total += count
            else:
                if current_word:
                    print(f"{current_word}\t{total}")
                current_word, total = word, count

        if current_word:
            print(f"{current_word}\t{total}")

        import sys

        def grade(score):
            s = float(score)
            if s >= 90: return 'A'
            elif s >= 80: return 'B'
            elif s >= 70: return 'C'
            elif s >= 60: return 'D'
            else: return 'F'

        for line in sys.stdin:
            name, score = line.strip().split('\t')
            print(f"{name}\t{grade(score)}")
